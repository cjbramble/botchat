Step 71–2 weeks

Python bot runner

Run 42 bots on 4–5 cheap OpenRouter models from one Python process, with a director that fixes how often anyone speaks and a daily dollar cap.

One Python process runs all 42 bots. Each bot has its own chat token and socket, but no bot decides when to speak. A director hands out turns at a fixed rate per channel, so your LLM bill is set by config, not by how chatty the models feel.

Cost is controlled in four layers, from inside out:

Layer Where What it caps
Director turn rate Bot runner Messages per channel per hour, so LLM calls per day are fixed by config.
Daily dollar budget Bot runner Sums the cost OpenRouter returns with every response; at the cap, every bot pauses until midnight UTC.
Server-side rate limit Chat (Step 2) In topic channels, 1 message per bot per 30 s and 1 per channel per 5 s (looser in #games), even if the runner has a bug.
Key credit limit OpenRouter dashboard A spending cap on the API key itself; requests fail with HTTP 402 when it's spent.

  • uv: projects, uv add, uv run.
  • asyncio: tasks, TaskGroup, wait_for.
  • websockets: the websockets.asyncio.client API.
  • OpenRouter quickstart (works with the OpenAI SDK by changing base_url), usage accounting (cost in every response) and limits (per-key credit caps, :free model limits).
  • The token bucket idea behind the director.

7.3 The Phoenix channel client (full code)

Phoenix's V2 JSON protocol sends every frame as a 5-element array: [join_ref, ref, topic, event, payload]. This client joins a list of topics, sends a heartbeat every 30 s, and reconnects with jittered backoff.

bots/phx.py
"""A minimal Phoenix Channels client (V2 JSON serializer) on top of websockets."""
import asyncio
import itertools
import json
import logging
import random
from collections.abc import Awaitable, Callable

from websockets.asyncio.client import connect
from websockets.exceptions import ConnectionClosed

log = logging.getLogger(__name__)
Handler = Callable[[str, str, dict], Awaitable[None]]  # (topic, event, payload)


class PhoenixClient:
    def __init__(self, base_url: str, token: str, topics: list[str], on_event: Handler):
        self.url = f"{base_url}/socket/websocket?vsn=2.0.0&token={token}"
        self.topics = topics
        self.on_event = on_event
        self.refs = itertools.count(1)
        self.ws = None

    async def push(self, topic: str, event: str, payload: dict) -> None:
        if self.ws is None:
            raise ConnectionError("not connected")
        join_ref = self.join_refs.get(topic)
        frame = [join_ref, str(next(self.refs)), topic, event, payload]
        await self.ws.send(json.dumps(frame))

    async def run_forever(self) -> None:
        """Connect, join every topic, dispatch events; reconnect with backoff on failure."""
        backoff = 1.0
        while True:
            try:
                async with connect(self.url) as ws:
                    self.ws = ws
                    self.join_refs = {}
                    for topic in self.topics:
                        ref = str(next(self.refs))
                        self.join_refs[topic] = ref
                        await ws.send(json.dumps([ref, ref, topic, "phx_join", {}]))
                    backoff = 1.0
                    heartbeat = asyncio.create_task(self._heartbeat())
                    try:
                        async for raw in ws:
                            _join_ref, _ref, topic, event, payload = json.loads(raw)
                            await self.on_event(topic, event, payload)
                    finally:
                        heartbeat.cancel()
            except (OSError, ConnectionClosed) as exc:
                log.warning("socket dropped (%s); reconnecting in %.0fs", exc, backoff)
            self.ws = None
            await asyncio.sleep(backoff + random.uniform(0, backoff))  # jitter: 42 bots mustn't reconnect in sync
            backoff = min(backoff * 2, 60)

    async def _heartbeat(self) -> None:
        while True:
            await asyncio.sleep(30)
            await self.ws.send(json.dumps([None, str(next(self.refs)), "phoenix", "heartbeat", {}]))

The jitter in the reconnect delay matters: when chat restarts, 42 bots reconnecting in the same millisecond is a small thundering herd.

7.4 The OpenRouter client (full code)

bots/llm.py
"""One shared OpenRouter client. OpenRouter speaks the OpenAI chat-completions API."""
import os
from dataclasses import dataclass

from openai import AsyncOpenAI

_client = None


def client() -> AsyncOpenAI:
    global _client
    if _client is None:
        _client = AsyncOpenAI(
            base_url="https://openrouter.ai/api/v1",
            api_key=os.environ["OPENROUTER_API_KEY"],
            timeout=15,
            max_retries=2,
        )
    return _client


@dataclass
class Reply:
    text: str
    cost_usd: float
    tokens: int


async def complete(model: str, system: str, lines: list[str], max_tokens: int) -> Reply:
    resp = await client().chat.completions.create(
        model=model,
        max_tokens=max_tokens,
        messages=[
            {"role": "system", "content": system},
            {"role": "user", "content": "\n".join(lines) or "(the channel is quiet)"},
        ],
    )
    usage = resp.usage
    cost = getattr(usage, "cost", None) or 0.0  # OpenRouter adds `cost` (USD) to usage
    text = (resp.choices[0].message.content or "").strip()
    return Reply(text=text, cost_usd=float(cost), tokens=getattr(usage, "total_tokens", 0) or 0)

7.5 The budget (full code)

bots/budget.py
"""A daily dollar budget shared by every bot. Resets at midnight UTC."""
import asyncio
from datetime import datetime, timedelta, timezone
from collections.abc import Callable

Clock = Callable[[], datetime]


def utc_now() -> datetime:
    return datetime.now(timezone.utc)


class Budget:
    def __init__(self, daily_usd: float, clock: Clock = utc_now):
        self.daily_usd = daily_usd
        self.clock = clock
        self.day = clock().date()
        self.spent = 0.0

    def _roll(self) -> None:
        today = self.clock().date()
        if today != self.day:
            self.day, self.spent = today, 0.0

    def charge(self, usd: float) -> None:
        self._roll()
        self.spent += usd

    def exhausted(self) -> bool:
        self._roll()
        return self.spent >= self.daily_usd

    def seconds_until_reset(self) -> float:
        now = self.clock()
        tomorrow = datetime.combine(now.date() + timedelta(days=1), datetime.min.time(), timezone.utc)
        return (tomorrow - now).total_seconds()

    async def wait_for_reset(self) -> None:
        await asyncio.sleep(self.seconds_until_reset() + 1)

The clock is injected so tests can jump to midnight instead of waiting for it.

7.6 The director

The director runs one loop per channel. Each loop sleeps about 3600 / per_hour seconds, checks the budget, picks one bot, and lets it speak once.

Exercise: pick_speaker and run_channel

Write bots/director.py with:

  • pick_speaker(bots, channel, recent_lines, exclude=None, rng=random): choose among bots whose home includes the channel, never exclude (the last speaker), and three times as likely if the bot's nick appears in the last 5 lines.
  • async run_channel(channel, per_hour, bots, budget, history, rng=random, sleep=asyncio.sleep): loop forever. Sleep interval × uniform(0.7, 1.3); if the budget is exhausted, wait for the reset; otherwise pick a speaker, await speaker.speak(channel), and charge the returned cost. One failed LLM call must not stop the loop.

Hint: weighted choice

random.choices(population, weights=[...], k=1)[0] picks one item with the given relative weights.

Solution 
def pick_speaker(bots, channel: str, recent_lines: list[str], exclude=None, rng=random):
    """Choose who talks next: someone who lives in this channel, not the last speaker,
    three times as likely if they were mentioned in the last 5 lines."""
    candidates = [b for b in bots if channel in b.home and b is not exclude]
    if not candidates:
        return None
    tail = " ".join(recent_lines[-5:]).lower()
    weights = [3 if b.nick.lower() in tail else 1 for b in candidates]
    return rng.choices(candidates, weights=weights, k=1)[0]


async def run_channel(channel: str, per_hour: int, bots, budget: Budget, history, rng=random, sleep=asyncio.sleep):
    """Loop forever: wait ~3600/per_hour seconds, then let one bot speak."""
    if per_hour <= 0:
        return
    interval = 3600 / per_hour
    last = None
    while True:
        await sleep(interval * rng.uniform(0.7, 1.3))  # jitter so channels don't tick in lockstep
        if budget.exhausted():
            log.warning("daily budget spent; pausing #%s until midnight UTC", channel)
            await budget.wait_for_reset()
            continue
        speaker = pick_speaker(bots, channel, history(channel), exclude=last, rng=rng)
        if speaker is None:
            continue
        try:
            budget.charge(await speaker.speak(channel))
        except Exception:  # one bad LLM call must not stop the channel
            log.exception("%s failed to speak in #%s", speaker.nick, channel)
        last = speaker

7.7 A bot

bots/bot.py
"""One AI participant: a persona, a model, a memory of each channel, and a socket."""
import asyncio
import logging
import re
from collections import defaultdict, deque
from dataclasses import dataclass, field

from . import llm

log = logging.getLogger(__name__)

RULES = (
    "You are chatting in an IRC channel. Reply with ONE short line (under 200 characters), "
    "no markdown, no lists, no greetings unless greeted. Stay on the channel topic and in character. "
    "React to what others just said; don't repeat yourself."
)


@dataclass
class Bot:
    nick: str
    persona: str
    model: str
    home: set[str]
    max_tokens: int = 120
    topics: dict[str, str] = field(default_factory=dict)
    history: dict[str, deque] = field(default_factory=lambda: defaultdict(lambda: deque(maxlen=30)))
    match: dict = field(default_factory=dict)  # latest match_state from #games
    client: object = None  # PhoenixClient, set by main
    budget: object = None  # Budget, set by main

    # ---- incoming events -------------------------------------------------
    async def on_event(self, topic: str, event: str, payload: dict) -> None:
        channel = topic.removeprefix("room:")
        if event == "phx_reply" and payload.get("status") == "ok":
            topic_text = payload.get("response", {}).get("topic")
            if topic_text:
                self.topics[channel] = topic_text
        elif event == "match_state":
            self.match = payload
        elif event == "new_msg":
            self.history[channel].append(f"<{payload['nick']}> {payload['body']}")
            if payload["nick"] == "GameServ" and f"{self.nick}, you're up" in payload["body"]:
                asyncio.create_task(self.take_turn(channel))

    # ---- speaking (only when the director says so) ------------------------
    async def speak(self, channel: str) -> float:
        system = f"{self.persona}\nChannel #{channel}: {self.topics.get(channel, '')}\n{RULES}"
        reply = await llm.complete(self.model, system, list(self.history[channel])[-15:], self.max_tokens)
        line = reply.text.splitlines()[0][:300] if reply.text else ""
        if line:
            await self.client.push(f"room:{channel}", "new_msg", {"body": line})
        log.info("spoke bot=%s model=%s channel=%s tokens=%d cost=%.6f",
                 self.nick, self.model, channel, reply.tokens, reply.cost_usd)
        return reply.cost_usd

on_event never calls the LLM by itself. Chat lines only update memory; the director decides when speak runs. The one exception is a game turn, which GameServ triggers.

Exercise: game turns

Write fallback_guess() and take_turn(channel):

  • The fallback is the midpoint of the known range, from the latest match_state (lo, hi; default 1–100).
  • If the budget allows, ask the model for one integer with max_tokens=5, wrapped in asyncio.wait_for(..., timeout=5). Charge its cost.
  • Use the model's number only if it parses and is inside the range; otherwise, or on any error or timeout, use the fallback.
  • Post the guess with self.client.push(f"room:{channel}", "new_msg", {"body": str(guess)}).
Solution 
    def fallback_guess(self) -> int:
        lo, hi = self.match.get("lo", 1), self.match.get("hi", 100)
        return (lo + hi) // 2

    async def take_turn(self, channel: str) -> None:
        guess = self.fallback_guess()
        if self.budget is None or not self.budget.exhausted():
            lo, hi = self.match.get("lo", 1), self.match.get("hi", 100)
            system = (f"{self.persona}\nYou are playing Guess the Number. The number is between "
                      f"{lo} and {hi}. Reply with ONLY one integer.")
            try:
                reply = await asyncio.wait_for(
                    llm.complete(self.model, system, list(self.history[channel])[-10:], 5), timeout=5)
                if self.budget:
                    self.budget.charge(reply.cost_usd)
                m = re.search(r"-?\d+", reply.text)
                if m and lo <= int(m.group()) <= hi:
                    guess = int(m.group())
            except Exception:
                log.warning("%s: LLM turn failed, guessing the midpoint", self.nick)
        await self.client.push(f"room:{channel}", "new_msg", {"body": str(guess)})

7.8 Putting it together (full code)

bots/main.py
"""Entry point: load config, build 42 bots, run sockets and directors together."""
import asyncio
import logging
import os
import tomllib
from pathlib import Path

from .bot import Bot
from .budget import Budget
from .director import run_channel
from .phx import PhoenixClient


def load_bots(config: dict, personas: list[dict]) -> list[Bot]:
    """Assign personas to model groups in order: the first `bots` personas get the first model, and so on."""
    models = [g["model"] for g in config["model_groups"] for _ in range(g["bots"])]
    if len(models) != len(personas):
        raise SystemExit(f"{len(personas)} personas but model_groups add up to {len(models)} bots")
    max_tokens = config["budget"]["max_output_tokens"]
    return [
        Bot(nick=p["nick"], persona=p["persona"], model=model, home=set(p["home"]), max_tokens=max_tokens)
        for p, model in zip(personas, models)
    ]


async def main() -> None:
    logging.basicConfig(level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
    root = Path(__file__).parent
    config = tomllib.loads((root / "config.toml").read_text())
    personas = tomllib.loads((root / "personas.toml").read_text())["bots"]
    bots = load_bots(config, personas)
    budget = Budget(config["budget"]["daily_usd"])
    chat_url = os.environ.get("CHAT_URL", "ws://localhost:4001")

    def history(channel: str) -> list[str]:  # any member's view of the channel will do
        for b in bots:
            if b.history.get(channel):
                return list(b.history[channel])
        return []

    async with asyncio.TaskGroup() as tg:
        for b in bots:
            b.budget = budget
            token = os.environ[f"BOT_TOKEN_{b.nick.upper()}"]
            b.client = PhoenixClient(chat_url, token, [f"room:{c}" for c in sorted(b.home)], b.on_event)
            tg.create_task(b.client.run_forever())
        for ch in config["channels"]:
            members = [b for b in bots if ch["name"] in b.home]
            tg.create_task(run_channel(ch["name"], ch["per_hour"], members, budget, history))


if __name__ == "__main__":
    asyncio.run(main())
